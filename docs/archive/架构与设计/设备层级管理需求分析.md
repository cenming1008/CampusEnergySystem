# 设备层级管理需求分析与解决方案

## 📋 问题分析

### 当前系统的不足

**现状：**
```python
class Device(SQLModel, table=True):
    id: int
    name: str          # 如："1309电表"
    location: str      # 如："A栋3单元1309"  ❌ 只是一个字符串！
    # ...
```

**问题：**

```
❌ 无法表示层级关系
   - location 只是一个字符串
   - 无法查询"A栋下所有设备"
   - 无法查询"3单元下所有设备"
   - 无法统计"1309房间的总能耗"

❌ 无法灵活筛选
   - 想查"A栋所有电表" → 需要模糊查询 location LIKE 'A栋%'
   - 想查"3单元所有水表" → 查询困难
   - 想统计每栋楼能耗 → 需要解析字符串

❌ 数据不一致
   - 可能写成 "A栋3单元1309"
   - 也可能写成 "A栋-3单元-1309"
   - 还可能写成 "A栋/3单元/1309"
   - 甚至写成 "1309室"（缺少楼栋信息）

❌ 无法管理组织结构
   - 楼栋、单元、房间都是隐式的
   - 无法单独管理
   - 无法添加额外信息（如建筑面积、负责人等）
```

## ✅ 解决方案

### 方案对比

| 方案 | 优点 | 缺点 | 推荐 |
|------|------|------|------|
| 方案1：字符串路径 | 简单 | ❌ 无法查询、统计 | ❌ 不推荐 |
| 方案2：JSON字段 | 灵活 | ❌ 查询困难、性能差 | ❌ 不推荐 |
| 方案3：专门表 | ✅ 结构清晰、查询高效 | 稍复杂 | ✅ 推荐 |

### 推荐方案：三表结构

```
┌─────────────────┐
│   Location      │  物理位置表（楼栋-单元-房间）
│   (位置层级)    │
└────────┬────────┘
         │ 1对多
         ↓
┌─────────────────┐
│   Device        │  设备表
│   (设备信息)    │
└────────┬────────┘
         │ 多对多
         ↓
┌─────────────────┐
│  DeviceGroup    │  设备分组表（逻辑分组）
│  (业务分组)     │
└─────────────────┘
```

## 🔧 技术实现

### 1. Location 表（物理位置层级）

```python
class LocationType(str, Enum):
    """位置类型枚举"""
    BUILDING = "building"    # 楼栋
    UNIT = "unit"           # 单元
    FLOOR = "floor"         # 楼层
    ROOM = "room"           # 房间
    WORKSHOP = "workshop"   # 车间
    AREA = "area"           # 区域

class Location(SQLModel, table=True):
    """位置表（支持层级结构）"""
    __tablename__ = "location"
    
    id: Optional[int] = Field(default=None, primary_key=True)
    
    # 层级关系
    parent_id: Optional[int] = Field(
        default=None, 
        foreign_key="location.id",
        description="父级位置ID（NULL表示顶级）"
    )
    
    # 位置信息
    name: str = Field(description="位置名称（如：A栋、3单元、1309）")
    location_type: str = Field(index=True, description="位置类型")
    code: Optional[str] = Field(default=None, unique=True, description="位置编码")
    
    # 完整路径（冗余字段，提升查询性能）
    full_path: Optional[str] = Field(
        default=None, 
        description="完整路径（如：/A栋/3单元/1309）"
    )
    
    # 额外信息
    description: Optional[str] = Field(default=None, description="描述")
    area_sqm: Optional[float] = Field(default=None, description="面积（平方米）")
    manager: Optional[str] = Field(default=None, description="负责人")
    
    # 状态
    is_active: bool = Field(default=True, description="是否启用")
    
    # 时间
    created_at: datetime = Field(default_factory=datetime.now)
    updated_at: datetime = Field(default_factory=datetime.now)
```

### 2. DeviceGroup 表（逻辑分组）

```python
class DeviceGroup(SQLModel, table=True):
    """设备分组表（用于业务逻辑分组）"""
    __tablename__ = "device_group"
    
    id: Optional[int] = Field(default=None, primary_key=True)
    name: str = Field(index=True, description="分组名称")
    code: Optional[str] = Field(default=None, unique=True, description="分组编码")
    description: Optional[str] = Field(default=None, description="分组描述")
    
    # 分组类型
    group_type: Optional[str] = Field(
        default=None, 
        description="分组类型（production/office/critical/backup）"
    )
    
    # 层级关系（可选）
    parent_id: Optional[int] = Field(
        default=None,
        foreign_key="device_group.id",
        description="父分组ID"
    )
    
    is_active: bool = Field(default=True)
    created_at: datetime = Field(default_factory=datetime.now)

class DeviceGroupMembership(SQLModel, table=True):
    """设备-分组关联表（多对多）"""
    __tablename__ = "device_group_membership"
    
    device_id: int = Field(primary_key=True, foreign_key="device.id")
    group_id: int = Field(primary_key=True, foreign_key="device_group.id")
    joined_at: datetime = Field(default_factory=datetime.now)
```

## 📊 使用示例

### 示例 1：创建位置层级

```python
from app.services.location_service import LocationService

# 创建 A栋
building_a = LocationService.create_location(
    session=session,
    name="A栋",
    location_type=LocationType.BUILDING,
    code="BUILD-A",
    area_sqm=5000.0,
    manager="张三"
)

# 创建 3单元（A栋下）
unit_3 = LocationService.create_location(
    session=session,
    name="3单元",
    location_type=LocationType.UNIT,
    code="BUILD-A-UNIT-3",
    parent_id=building_a.id
)

# 创建 1309房间（3单元下）
room_1309 = LocationService.create_location(
    session=session,
    name="1309",
    location_type=LocationType.ROOM,
    code="BUILD-A-UNIT-3-ROOM-1309",
    parent_id=unit_3.id,
    area_sqm=80.0
)

# 创建设备并关联位置
device = DeviceService.create_device_smart(
    session=session,
    name="1309电表",
    sn="METER-1309",
    device_type="load",
    location_id=room_1309.id  # 🆕 关联到1309房间
)

print(f"设备位置：{device.full_location_path}")
# 输出：A栋 > 3单元 > 1309
```

## 🔄 迁移策略

### 步骤 1：解析现有 location 字段

```python
def migrate_locations():
    """从现有的 location 字符串迁移到层级结构"""
    
    # 解析现有设备的 location 字段
    devices = session.exec(select(Device)).all()
    
    for device in devices:
        if not device.location:
            continue
        
        # 解析 "A栋3单元1309" → ["A栋", "3单元", "1309"]
        parts = parse_location_string(device.location)
        
        # 创建或获取位置层级
        location_id = ensure_location_hierarchy(session, parts)
        
        # 更新设备的 location_id
        device.location_id = location_id
        session.add(device)
    
    session.commit()
```

### 步骤 2：保持向后兼容

```python
class Device(SQLModel, table=True):
    location_id: Optional[int] = None  # 新字段
    location: Optional[str] = None      # 旧字段，保留
    
    @property
    def full_location_path(self) -> str:
        """获取完整位置路径（优先用新字段）"""
        if self.location_id:
            # 从 Location 表构建路径
            return build_location_path(self.location_id)
        else:
            # 使用旧字段
            return self.location or "未知位置"
```
