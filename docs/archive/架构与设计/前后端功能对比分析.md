// frontend/src/api/energy.ts
import request from '@/utils/request'

export interface EnergyData {
  device_id: number
  energy_type: string
  consumption: number
  flow_rate: number | null
  timestamp: string
}

export function getEnergyData(params: {
  energy_type?: string
  device_id?: number
  start_time?: string
  end_time?: string
  limit?: number
}) {
  return request.get<any, EnergyData[]>('/energy/data', { params })
}

export function getEnergyStatistics(params: {
  energy_type?: string
  start_time: string
  end_time: string
  group_by?: string
}) {
  return request.get('/energy/statistics', { params })
}
